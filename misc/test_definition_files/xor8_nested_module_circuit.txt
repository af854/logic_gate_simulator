MODULE XOR2
    INPUTS A, B;
    OUTPUTS Y;

    DEVICES
        XOR X;

    CONNECT
        X.I1 = A;
        X.I2 = B;
        Y = X;
ENDMODULE;


MODULE XOR4
    INPUTS A0, A1, A2, A3;
    OUTPUTS Y;

    DEVICES
        XOR2 L, R, O;

    CONNECT
        L.A = A0;
        L.B = A1;

        R.A = A2;
        R.B = A3;

        O.A = L.Y;
        O.B = R.Y;

        Y = O.Y;
ENDMODULE;


MODULE XOR8
